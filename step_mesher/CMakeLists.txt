cmake_minimum_required(VERSION 3.20)
project(surfsmooth3d_step VERSION 3.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(STEP_MESHER_BUILD_MEX "Build MATLAB MEX gateway" ON)
option(STEP_MESHER_BUILD_TESTS "Build native tests" ON)
option(STEP_MESHER_SANITIZERS "Instrument the mesher with ASan and UBSan" OFF)
set(STEP_MESHER_DEPENDENCY_PREFIX "" CACHE PATH "V3's independent common Gmsh/OCCT installation")
if(NOT STEP_MESHER_DEPENDENCY_PREFIX)
  message(FATAL_ERROR "Set STEP_MESHER_DEPENDENCY_PREFIX to the independent Gmsh 4.12.2 / OCCT 7.9.3 installation. The Python-framework Gmsh binary is incompatible with native sharing.")
endif()
get_filename_component(STEP_MESHER_DEPENDENCY_PREFIX "${STEP_MESHER_DEPENDENCY_PREFIX}" REALPATH)
set(GMSH_INCLUDE_DIR "${STEP_MESHER_DEPENDENCY_PREFIX}/include" CACHE PATH "Gmsh header" FORCE)
find_library(GMSH_LIBRARY NAMES gmsh PATHS "${STEP_MESHER_DEPENDENCY_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
find_package(Gmsh REQUIRED)
find_package(OpenCASCADE 7.9.3 EXACT CONFIG REQUIRED PATHS "${STEP_MESHER_DEPENDENCY_PREFIX}/lib/cmake/opencascade" "${STEP_MESHER_DEPENDENCY_PREFIX}/cmake" NO_DEFAULT_PATH)
foreach(_dep IN ITEMS "${GMSH_LIBRARY}" "${OpenCASCADE_INCLUDE_DIR}")
  get_filename_component(_real "${_dep}" REALPATH)
  string(FIND "${_real}/" "${STEP_MESHER_DEPENDENCY_PREFIX}/" _under)
  if(NOT _under EQUAL 0)
    message(FATAL_ERROR "Native pointer ABI requires the selected shared prefix: ${_real}")
  endif()
endforeach()
set(_occ_libraries TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKXSBase TKDESTEP)
set(_runtime_rpath "")
if(APPLE)
  list(APPEND _runtime_rpath "@loader_path" "@loader_path/../lib" "@loader_path/../../../../lib")
else()
  list(APPEND _runtime_rpath "$ORIGIN" "$ORIGIN/../lib" "$ORIGIN/../../../../lib")
endif()
list(APPEND _runtime_rpath "${STEP_MESHER_DEPENDENCY_PREFIX}/lib" "${CMAKE_INSTALL_PREFIX}/lib")
set(CMAKE_BUILD_RPATH "${CMAKE_BINARY_DIR}/lib;${STEP_MESHER_DEPENDENCY_PREFIX}/lib")
set(CMAKE_INSTALL_RPATH "${_runtime_rpath}")
add_library(step_mesher_core SHARED cpp/src/mesher.cpp cpp/src/sample_nodes.cpp cpp/src/occ_context.cpp)
target_include_directories(step_mesher_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/cpp/include" PRIVATE "${OpenCASCADE_INCLUDE_DIR}")
target_link_libraries(step_mesher_core PUBLIC Gmsh::Gmsh PRIVATE ${_occ_libraries} ${CMAKE_DL_LIBS})
set_target_properties(step_mesher_core PROPERTIES OUTPUT_NAME surfsmooth3d_step_core VERSION 3.0.0 SOVERSION 3 LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
target_compile_definitions(step_mesher_core PRIVATE STEP_MESHER_DEPENDENCY_PREFIX="${STEP_MESHER_DEPENDENCY_PREFIX}" STEP_MESHER_RUNTIME_VERSION="${PROJECT_VERSION}")
if(STEP_MESHER_SANITIZERS)
  target_compile_options(step_mesher_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(step_mesher_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(step_mesher_cli cpp/src/cli.cpp)
target_link_libraries(step_mesher_cli PRIVATE step_mesher_core)
# Keep the package together; configuration/build never replaces source binaries.
file(GLOB_RECURSE _matlab_files RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../matlab" "${CMAKE_CURRENT_SOURCE_DIR}/../matlab/*.m")
foreach(_file IN LISTS _matlab_files)
  configure_file("${CMAKE_CURRENT_SOURCE_DIR}/../matlab/${_file}" "${CMAKE_BINARY_DIR}/matlab/${_file}" COPYONLY)
endforeach()
add_executable(step_mesher_occ_info cpp/src/occ_info.cpp)
target_link_libraries(step_mesher_occ_info PRIVATE step_mesher_core)
set_target_properties(step_mesher_occ_info PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/matlab/+surfsmooth3d/+stepmesher/private")
if(STEP_MESHER_BUILD_MEX)
  find_package(Matlab REQUIRED COMPONENTS MEX_COMPILER MX_LIBRARY)
  matlab_add_mex(NAME step_mesher_mex SRC mex/step_mesher_mex.cpp LINK_TO step_mesher_core)
  set_target_properties(step_mesher_mex PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/matlab/+surfsmooth3d/+stepmesher/private" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/matlab/+surfsmooth3d/+stepmesher/private")
  install(TARGETS step_mesher_mex LIBRARY DESTINATION matlab/+surfsmooth3d/+stepmesher/private RUNTIME DESTINATION matlab/+surfsmooth3d/+stepmesher/private)
endif()
install(TARGETS step_mesher_core step_mesher_cli LIBRARY DESTINATION lib RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
install(TARGETS step_mesher_occ_info RUNTIME DESTINATION matlab/+surfsmooth3d/+stepmesher/private)
install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../matlab/" DESTINATION matlab FILES_MATCHING PATTERN "*.m")
install(DIRECTORY cpp/include/ DESTINATION include)
if(STEP_MESHER_BUILD_TESTS)
  enable_testing()
  add_executable(step_mesher_smoke cpp/tests/smoke.cpp)
  target_link_libraries(step_mesher_smoke PRIVATE step_mesher_core)
  add_test(NAME cpp_smoke_import COMMAND step_mesher_smoke "${CMAKE_CURRENT_SOURCE_DIR}/examples/step_files/test_geometry_1.step")
  add_executable(test_occ_frozen cpp/tests/test_occ_frozen.cpp)
  target_link_libraries(test_occ_frozen PRIVATE step_mesher_core)
  add_executable(native_contract cpp/tests/native_contract.cpp)
  target_link_libraries(native_contract PRIVATE step_mesher_core)
  add_test(NAME native_contract COMMAND native_contract "${CMAKE_CURRENT_SOURCE_DIR}/examples/step_files/test_geometry_1.step" "$<TARGET_FILE:step_mesher_occ_info>")
  add_executable(test_occ_context cpp/tests/test_occ_context.cpp cpp/tests/occ_context_probe.cpp)
  add_executable(test_projection_reuse cpp/tests/test_projection_reuse.cpp)
  target_include_directories(test_projection_reuse PRIVATE "${OpenCASCADE_INCLUDE_DIR}")
  target_link_libraries(test_projection_reuse PRIVATE step_mesher_core ${_occ_libraries})
  add_test(NAME projection_reuse COMMAND test_projection_reuse
    "${CMAKE_CURRENT_SOURCE_DIR}/examples/step_files/UV_curved_bezier.step")
  add_test(NAME projection_reuse_failure_recovery COMMAND test_projection_reuse
    "${CMAKE_CURRENT_SOURCE_DIR}/examples/step_files/UV_curved_bezier.step" --failure-probe)
  set_tests_properties(projection_reuse_failure_recovery PROPERTIES TIMEOUT 20)
  add_executable(test_occ_import_equivalence cpp/tests/test_occ_import_equivalence.cpp)
  target_link_libraries(test_occ_import_equivalence PRIVATE step_mesher_core)
  file(GLOB _step_fixtures "${CMAKE_CURRENT_SOURCE_DIR}/examples/step_files/*.step")
  foreach(_fixture IN LISTS _step_fixtures)
    get_filename_component(_case "${_fixture}" NAME_WE)
    add_test(NAME "import_equivalence_${_case}" COMMAND test_occ_import_equivalence "${_fixture}")
  endforeach()
  add_executable(test_occ_retention cpp/tests/test_occ_retention.cpp)
  target_link_libraries(test_occ_retention PRIVATE step_mesher_core)
  set(_frozen_fixtures "${CMAKE_CURRENT_SOURCE_DIR}/../tests/step/fixtures/occ_operations")
  set(STEP_FIXTURE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/examples/step_files/UV_curved_bezier.step")
  foreach(_operation IN ITEMS cad-measurements helper-curves edge-samples feature-samples feature-nodes project-nodes)
    set(_request "${CMAKE_BINARY_DIR}/frozen-fixtures/${_operation}.request.txt")
    if(NOT _operation STREQUAL "cad-measurements")
      configure_file("${_frozen_fixtures}/${_operation}.request.in" "${_request}" @ONLY)
    endif()
    add_test(NAME "frozen_${_operation}" COMMAND "${CMAKE_COMMAND}"
      "-DOPERATION=${_operation}" "-DHELPER=$<TARGET_FILE:step_mesher_occ_info>"
      "-DFROZEN=$<TARGET_FILE:test_occ_frozen>" "-DSTEP_FILE=${STEP_FIXTURE_PATH}"
      "-DREQUEST=${_request}" "-DMAPPING=${_frozen_fixtures}/frozen-map.txt"
      "-DEXPECTED=${_frozen_fixtures}/${_operation}.expected.txt"
      "-DOUTPUT_DIR=${CMAKE_BINARY_DIR}/frozen-results/${_operation}"
      -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/CheckFrozenOperation.cmake")
    if(NOT _operation STREQUAL "cad-measurements")
      add_test(NAME "retention_${_operation}" COMMAND "${CMAKE_COMMAND}"
        "-DOPERATION=${_operation}" "-DRETENTION=$<TARGET_FILE:test_occ_retention>"
        "-DREQUEST=${_request}"
        "-DHELPER_REQUEST=${CMAKE_BINARY_DIR}/frozen-fixtures/helper-curves.request.txt"
        "-DOUTPUT_DIR=${CMAKE_BINARY_DIR}/retention-results/${_operation}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/CheckRetention.cmake")
    endif()
  endforeach()
  target_include_directories(test_occ_context PRIVATE "${OpenCASCADE_INCLUDE_DIR}")
  target_link_libraries(test_occ_context PRIVATE step_mesher_core ${_occ_libraries})
  add_test(NAME occ_context COMMAND test_occ_context "${CMAKE_CURRENT_SOURCE_DIR}/examples/step_files/test_geometry_1.step")
  if(STEP_MESHER_BUILD_MEX)
    matlab_add_mex(NAME occ_context_probe SRC cpp/tests/occ_context_probe_mex.cpp cpp/tests/occ_context_probe.cpp LINK_TO step_mesher_core ${_occ_libraries})
    target_include_directories(occ_context_probe PRIVATE "${OpenCASCADE_INCLUDE_DIR}")
  endif()
endif()
